// OpenSky Network API configuration var OPENSKY_API_BASE = 'https://cors-anywhere.herokuapp.com/https://opensky-api.org/api/states/all'; var AUCKLAND_LAT = -36.8862; var AUCKLAND_LNG = 174.7651; var SEARCH_RADIUS_KM = 100; // Convert kilometers to degrees (approximate) function kmToDegrees(km) { return km / 111.0; } var searchRadiusDeg = kmToDegrees(SEARCH_RADIUS_KM); // Calculate bounds for OpenSky API (lamin, lomin, lamax, lomax) var bounds = { laMin: AUCKLAND_LAT - searchRadiusDeg, loMin: AUCKLAND_LNG - searchRadiusDeg, laMax: AUCKLAND_LAT + searchRadiusDeg, loMax: AUCKLAND_LNG + searchRadiusDeg }; var flightMarkers = {}; // Fetch flights from OpenSky Network function loadFlights() { var url = OPENSKY_API_BASE + '?lamin=' + bounds.laMin + '&lomin=' + bounds.loMin + '&lamax=' + bounds.laMax + '&lomax=' + bounds.loMax; fetch(url) .then(response => { if (!response.ok) { throw new Error('OpenSky API returned status: ' + response.status); } return response.json(); }) .then(data => { // Process flight data if (!data || !data.states) { console.warn('Invalid OpenSky response'); return; } // Clear current flight markers for (var id in flightMarkers) { if (flightMarkers.hasOwnProperty(id)) { layers.flights.removeLayer(flightMarkers[id]); } } flightMarkers = {}; // OpenSky returns flights as arrays with the following fields: // [0] = ICAO24 (unique transponder code) // [1] = callsign // [2] = origin_country // [3] = time_position // [4] = last_contact // [5] = longitude // [6] = latitude // [7] = baro_altitude (barometric altitude in m) // [8] = on_ground (bool) // [9] = velocity (speed over ground in m/s) // [10] = true_track (heading in degrees) // [11] = vertical_rate // [12] = sensors // [13] = geo_altitude (geometric altitude) // [14] = squawk // [15] = spi (special purpose indicator) // [16] = position_source for (var i = 0; i < data.states.length; i++) { var state = data.states[i]; var icao24 = state[0] || 'Unknown'; var callsign = (state[1] || '').trim() || icao24; var origin = state[2] || 'Unknown'; var lon = state[5]; var lat = state[6]; var baroAltitude = state[7]; var geoAltitude = state[13]; var altitude = geoAltitude !== null ? geoAltitude : baroAltitude; var velocity = state[9]; var heading = state[10]; // Skip if coordinates are null if (lat === null || lon === null) continue; // Convert velocity from m/s to km/h var speedKmh = velocity !== null ? (velocity * 3.6).toFixed(1) : 'N/A'; // Convert altitude from meters to feet var altitudeFt = altitude !== null ? (altitude * 3.28084).toFixed(0) : 'N/A'; // Heading degrees (0-360) var headingDeg = heading !== null ? heading.toFixed(1) : 'N/A'; var tooltipHtml = '' + callsign + '
' + 'ICAO24: ' + icao24 + '
' + 'Origin: ' + origin + '
' + 'Speed: ' + speedKmh + ' km/h
' + 'Altitude: ' + altitudeFt + ' ft
' + 'Heading: ' + headingDeg + '°'; var flightMarker = L.marker([lat, lon], { icon: flightIcon, rotationAngle: heading || 0 }); flightMarker.bindTooltip(tooltipHtml); flightMarker.addTo(layers.flights); flightMarkers[icao24] = flightMarker; } console.log('Loaded ' + Object.keys(flightMarkers).length + ' flights from OpenSky Network'); }) .catch(err => { console.error('Error loading flights from OpenSky Network:', err); $("#status").text("Error loading flights: " + err.message); }); }
